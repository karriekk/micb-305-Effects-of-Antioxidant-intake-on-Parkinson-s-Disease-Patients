# Extract pathways that meet our significance thresholds (p-adjust < 0.05 & abs(log2fc) >= 1)

Load and Process Functional Pathway Data

Import raw pathway abundances, apply a CLR transformation, and subset for the significantly differentially abundant pathways identified in aim 3 for beta-carotene.

# Load raw pathway abundance table (rows = pathways, cols = samples)
pathway_counts = read.delim('../../QIIME2_outputs/path_abun_unstrat.tsv', row.names = 1)

# Manually remove PD patients with missing dietary antioxidant intake values and MoCA scores from the pathway abundance table
# The PD patient IDs are: F1452, F3436, F4446, F4484, F4660, F5362, F6222, F6413, F8334, F8922, F9523, F9556, F9856, F2501
pathway_counts[c("F1452", "F3436", "F4446", "F4484", "F4660", "F5362", "F6222", "F6413", "F8334", "F8922", "F9523", "F9556", "F9856", "F2501")] = NULL

# Calculate a pseudocount (half the minimum non-zero value across all the pathways)
min_nonzero = min(pathway_counts[pathway_counts > 0]) # 0.331428571428571

# Apply CLR transformation and transform the results into a data frame 
pathway_clr = clr(pathway_counts + min_nonzero/2) %>% as.data.frame()

# Subset the table for only significantly differentially abundant pathways identified in aim 3 and transpose so samples are rows
sig_pathways_beta_carotene = as.data.frame(t(pathway_clr[significant_pathways_BC, , drop = FALSE]))

# Calculate Spearman coefficient
spearman_BC = sig_pathways_beta_carotene %>% cor(method = "spearman") %>% 
  as.data.frame() %>% 
  rownames_to_column("PWY1") %>% 
  pivot_longer(cols = -PWY1, names_to = "PWY2", values_to = "cor") %>% 
  arrange(-cor)

# Select only those that are strongly correlated with a rho threshold of greater than 0.9
verycor_BC = spearman_BC %>% filter(cor > 0.9) %>% 
  filter(PWY1 != PWY2) # Remove self-correlations

# To avoid manually grouping them, since we have a lot of pathways that are highly correlated, we used a for loop approach:
# We will keep the pathway from each redundant pair that has the lowest p-value in Maaslin2.

# Create a lookup table for effect sizes from our Maaslin2 results
p_values_BC = significant_df_BC %>% select(feature, p_adjust) 

# Identify which pathways to drop
pathways_to_drop_BC = c()

for(i in 1:nrow(verycor_BC)) {
  p1 = verycor_BC$PWY1[i]
  p2 = verycor_BC$PWY2[i]
  
  # Only process if we haven't already dropped one of them in a previous loop
  if(!(p1 %in% pathways_to_drop_BC) && !(p2 %in% pathways_to_drop_BC)){
    
    # Compare their p-values
    p_value_p1 = p_values_BC$p_adjust[p_values_BC$feature == p1]
    p_value_p2 = p_values_BC$p_adjust[p_values_BC$feature == p2]
    
    # Drop the one with the higher p-value
    if(p_value_p1 > p_value_p2) {
      pathways_to_drop_BC = c(pathways_to_drop_BC, as.character(p1))
    } else {
      pathways_to_drop_BC = c(pathways_to_drop_BC, as.character(p2))
    }
  }
}

pathways_to_drop_BC = unique(pathways_to_drop_BC)

# Create our final, non-redundant list
final_pathways_BC = setdiff(significant_pathways_BC, pathways_to_drop_BC)
# Started with 21 pathways, kept 10 non-redundant pathways.

# Update our working matrix to only include the non-redundant pathways
sig_pathways_beta_carotene_filtered = sig_pathways_beta_carotene %>% select(all_of(final_pathways_BC))

PCoA analysis

Use Aitchison distance to run PCoA analysis and visualize using ggplot2.

# Compute dissimilarity matrix - Aitchison distance on CLR data
aitchison_dist_BC = dist(sig_pathways_beta_carotene_filtered, method = "euclidean")
# View(as.matrix(euclidean_dist_BC)) # Each pairwise comparison included

# Run PCoA analysis
pcoa_BC = cmdscale(aitchison_dist_BC, k = 2, eig = TRUE)

# Extract coordinates and variance explained
points_BC = as.data.frame(pcoa_BC$points)

colnames(points_BC) = c("PCoA1", "PCoA2")

points_BC$SampleID = rownames(points_BC)

points_BC = merge(
  points_BC,
  parkinsons_metadata_MoCA_NA_removed[, c("SampleID", "moca_total", "Sex", "Age", "entacapone", "MoCA_score_categorization")],
  by = "SampleID"
)

variance_explained_BC = round(100 * pcoa_BC$eig / sum(pcoa_BC$eig), 2)

# Visualization with ggplot2

PERMANOVA test

PERMANOVA test is performed to determine the statistical significance of the correlation.

# Match the Parkinson's metadata to the distance matrix sample order --> reorders metadata so rows match the distance matrix order exactly
parkinsons_metadata_match_BC = parkinsons_metadata_MoCA_NA_removed[match(labels(aitchison_dist_BC), # Gets the sample names from the distance matrix
                                                                         parkinsons_metadata_MoCA_NA_removed$SampleID), ]

# Remove rows with missing values (10 PD patients with missing entacapone intake information are removed)
parkinsons_metadata_match_BC = parkinsons_metadata_match_BC[complete.cases(parkinsons_metadata_match_BC[, c("moca_total", "MoCA_score_categorization", "Age", "Sex", "entacapone")]), ]

# Save sample IDs of kept samples
keep_BC = parkinsons_metadata_match_BC$SampleID

# Subset the distance matrix
dist_mat_BC = as.dist(as.matrix(aitchison_dist_BC)[ # Converts distance object into a full square matrix
  match(keep_BC, labels(aitchison_dist_BC)), 
  match(keep_BC, labels(aitchison_dist_BC)) 
])

# Run PERMANOVA test to assess whether variation in the pathway distance matrix is associated with MoCA score after accounting for age, sex, and entacapone use
set.seed(421)

permanova_test = adonis2(
  dist_mat_BC ~ MoCA_score_categorization + Age + Sex + entacapone,
  data = parkinsons_metadata_match_BC,
  by = "margin",
  permutations = 999
)
permanova_test # p-value for moca_total is 0.247
